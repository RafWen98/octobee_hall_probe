# name when the desktop icon fails to start. pyproject.toml is the source of
# truth for versions; these mirror it.
#
# Note that this file installs the DEPENDENCIES only. The application itself is
# a package under src/, and wants an editable install:
#
#   pip install -e ".[gui,dev]"          # what the bench and CI both use
#   pip install -e .                     # command-line tools, no GUI
#   pip install -r requirements.txt      # dependencies alone, no octobee
#
# -e is not optional. A plain `pip install .` copies the package into
# site-packages, and from then on the files being edited are not the files
# being run.
